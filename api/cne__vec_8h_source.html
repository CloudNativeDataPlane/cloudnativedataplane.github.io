94static __cne_always_inline void *
95_vec_realloc_data(void *vec, uint32_t nelem, uint32_t esize)
96{
97 vec_hdr_t *h;
98 uint32_t sz = (nelem * esize) + sizeof(vec_hdr_t);
99
100 h = (vec) ? vec_header(vec) : NULL;
101 h = realloc(h, sz);
102 if (!h)
103 return NULL;
104 if (!vec)
105 memset(h, 0, sz);
106 h->alen = nelem;
107 h->esize = esize;
108 return (void *)h->data;
109}

359static inline void
360vec_remove(void *v, uint32_t nb)
361{
362 vec_hdr_t *h = vec_header(v);
363
364 if (h && h->len && nb <= h->len) {
365 int n = h->len - nb;
366
367 if (n > 0)
368 memmove(v, CNE_PTR_ADD(v, (nb * h->esize)), n * h->esize);
369 h->len = n;
370 }
371}
