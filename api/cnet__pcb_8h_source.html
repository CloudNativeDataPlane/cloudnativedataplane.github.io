55static inline void
56cnet_pcb_free(struct pcb_entry *pcb)
57{
58 if (pcb) {
59 memset(pcb, 0, sizeof(struct pcb_entry));
60 pcb->closed = 1;
61 pcb->ip_proto = -1;
62 mempool_put(this_stk->pcb_objs, (void *)pcb);
63 }
64}

66static inline struct pcb_entry *
67cnet_pcb_alloc(struct pcb_hd *hd, uint16_t proto)
68{
69 struct pcb_entry *pcb;
70
71 if (mempool_get(this_stk->pcb_objs, (void *)&pcb) < 0)
72 return NULL;
73
74 pcb->closed = 0;
75 pcb->ip_proto = proto;
76
77 vec_add(hd->vec, pcb);
78
79 return pcb;
80}

82static inline void
83cnet_pcb_delete(struct pcb_hd *hd, struct pcb_entry *pcb)
84{
85 struct pcb_entry *p;
86
87 vec_foreach_ptr (p, hd->vec) {
88 if (p == pcb) {
89 cnet_pcb_free(pcb);
