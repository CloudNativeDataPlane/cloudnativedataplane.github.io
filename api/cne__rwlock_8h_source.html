64static inline void
65cne_rwlock_read_lock(cne_rwlock_t *rwl)
66{
67 int32_t x;
68 int success = 0;
69
70 while (success == 0) {
71 x = __atomic_load_n(&rwl->cnt, __ATOMIC_RELAXED);
72 /* write lock is held */
73 if (x < 0) {
74 cne_pause();
75 continue;
76 }
77 success = __atomic_compare_exchange_n(&rwl->cnt, &x, x + 1, 1, __ATOMIC_ACQUIRE,
78 __ATOMIC_RELAXED);
79 }
80}
