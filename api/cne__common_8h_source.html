795static inline uint64_t
796cne_str_to_size(const char *str)
797{
798 char *endptr;
799 unsigned long long size;
800
801 while (isspace((int)*str))
802 str++;
803 if (*str == '-')
804 return 0;
805
806 errno = 0;
807 size = strtoull(str, &endptr, 0);
808 if (errno)
809 return 0;
810
811 if (*endptr == ' ')
812 endptr++; /* allow 1 space gap */
813
814 switch (*endptr) {
815 case 'G':
816 case 'g':
817 size *= 1024; /* fall-through */
818 case 'M':
819 case 'm':
820 size *= 1024; /* fall-through */
821 case 'K':
822 case 'k':
823 size *= 1024; /* fall-through */
824 default:
825 break;
826 }
827 return size;
828}
