30/*
31 * protoFuncs_t - structure to hold all of the protocol function pointers.
32 *
33 * Each function pointer must have a valid value, which means may have to set
34 * the function pointer to the nullProtocol routine, if not supported.
35 *
36 * The following functions are called as indicated below.
37 */
38typedef int (*close_func_t)(struct chnl *ch);
39typedef int (*send_func_t)(struct chnl *ch, pktmbuf_t **mbufs, uint16_t nb_mbufs);
40typedef int (*recv_func_t)(struct chnl *ch, pktmbuf_t **mbufs, int nb_mbufs);
41typedef int (*bind_func_t)(struct chnl *ch, struct in_caddr *pAddr, int len);
42typedef int (*connect_func_t)(struct chnl *ch, struct in_caddr *to, int slen);
43typedef int (*shutdown_func_t)(struct chnl *ch, int how);
44typedef int (*accept_func_t)(struct chnl *ch, struct in_caddr *addr, int *addrlen);
45typedef int (*listen_func_t)(struct chnl *ch, int backlog);
46
47struct proto_funcs {
48 close_func_t close_func;
49 recv_func_t recv_func;
50 send_func_t send_func;
51 bind_func_t bind_func;
52 connect_func_t connect_func;
53 shutdown_func_t shutdown_func;
54 accept_func_t accept_func;
55 listen_func_t listen_func;
56};
